-simple pass/fail
proc check_ci {} {
    set repo "M-J-A-Baig/eFPGA"

    puts "Checking latest GitHub Actions run..."

    set run_id [exec gh run list --repo $repo --branch main --limit 1 --json databaseId --jq {.[0].databaseId}]

    if {$run_id eq ""} {
        puts "No GitHub Actions runs found."
        return
    }

    puts "Watching CI run: $run_id"

    if {[catch {
        exec gh run watch $run_id --repo $repo --exit-status
    } result]} {
        puts "CI FAILED or could not be checked."
        puts $result
    } else {
        puts "CI PASSED!"
    }
}




check_ci
